async function order(){ if(!cart.length){ msg("Add item first"); return; } const total=cart.reduce((a,x)=>a+x.price,0); const customer_name=document.getElementById("name").value.trim(); const phone=document.getElementById("phone").value.trim(); const address=document.getElementById("address").value.trim(); const pincode=document.getElementById("pin").value.trim(); const payment_method=document.getElementById("pay").value; if(!customer_name||!phone||!address||!pincode){ msg("Please complete all details"); return; } try{ const r=await fetch("/api/orders",{ method:"POST", headers:{"Content-Type":"application/json"}, body:JSON.stringify({ customer_name, phone, address, pincode, payment_method, total }) }); const j=await r.json(); if(r.ok && j.id){ msg("Order placed successfully! Order ID: "+j.id); cart=[]; renderCart(); }else{ msg(j.error||"Order failed"); } }catch(e){ msg("Unable to place order. Please try again."); } }
